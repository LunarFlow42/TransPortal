cmake_minimum_required(VERSION 3.20)
project(transportal_kwin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

enable_testing()
add_executable(gesture_test gesture_test.cpp)
add_test(NAME gesture_state_machine COMMAND gesture_test)

option(TRANSPORTAL_GESTURE_TEST_ONLY "Build just the portable gesture tests" OFF)
if(NOT TRANSPORTAL_GESTURE_TEST_ONLY)
    find_package(ECM REQUIRED NO_MODULE)
    list(APPEND CMAKE_MODULE_PATH "${ECM_MODULE_PATH}")
    find_package(Qt6 REQUIRED COMPONENTS DBus Widgets)
    find_package(KWin 6.3 REQUIRED)
    include(GNUInstallDirs)
    add_library(transportal-gestures MODULE plugin.cpp)
    set_target_properties(transportal-gestures PROPERTIES AUTOMOC ON PREFIX "")
    target_include_directories(transportal-gestures PRIVATE compat)
    target_link_libraries(transportal-gestures PRIVATE KWin::kwin Qt6::DBus)
    option(TRANSPORTAL_KWIN_SMOKE_TESTS "Compile test-only synthetic input entrypoint into the smoke-test plugin" OFF)
    if(TRANSPORTAL_KWIN_SMOKE_TESTS)
        target_compile_definitions(transportal-gestures PRIVATE TRANSPORTAL_KWIN_SMOKE_TESTS=1)
        message(STATUS "Test-only synthetic input enabled: this plugin has no install target")
    else()
        set(TRANSPORTAL_QT_PLUGINDIR "${CMAKE_INSTALL_LIBDIR}/qt6/plugins" CACHE STRING "Qt 6 plugin directory")
        install(TARGETS transportal-gestures LIBRARY DESTINATION "${TRANSPORTAL_QT_PLUGINDIR}/kwin/plugins")
    endif()
endif()
